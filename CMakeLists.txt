cmake_minimum_required(VERSION 3.24)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" WOODY_VERSION)
string(STRIP "${WOODY_VERSION}" WOODY_VERSION)
if(NOT WOODY_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
    message(FATAL_ERROR "VERSION must contain a version in major.minor.patch format")
endif()
project(Woody VERSION "${WOODY_VERSION}" LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
if(MSVC)
    add_compile_options(/utf-8)
endif()
include(CTest)
add_subdirectory(qt)
